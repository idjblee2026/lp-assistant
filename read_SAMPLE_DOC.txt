import os
import docx

# 워드 폴더 경로
WORD_DIR = r"D:\LP_Agent\word_reports"

# 첫 번째 워드 파일 선택
word_files = [f for f in os.listdir(WORD_DIR) if f.endswith(".docx") and not f.startswith("~$")]

if word_files:
    sample_file = word_files[0]
    file_path = os.path.join(WORD_DIR, sample_file)
    
    print("=" * 50)
    print(f"선택된 테스트 파일: {sample_file}")
    print("=" * 50)
    
    # 워드 문서 열기
    doc = docx.Document(file_path)
    
    # 본문 문단 읽어서 출력
    full_text = []
    for para in doc.paragraphs:
        if para.text.strip():  # 빈 줄 제외
            full_text.append(para.text)
            
    print("\n[워드 문서 내용 미리보기 (첫 5줄)]")
    for line in full_text[:5]:
        print(f"> {line}")
        
    print(f"\n- 전체 문단 수: {len(full_text)}개")
    print("=" * 50)
else:
    print("워드 파일을 찾을 수 없습니다.")
